cmake_minimum_required(VERSION 3.8)
project(robotx_task1_nav)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ompl REQUIRED)
foreach(dep rclcpp rclcpp_lifecycle nav2_core nav2_costmap_2d nav2_util dwb_core dwb_critics dwb_msgs nav_2d_msgs pluginlib robotx_task1_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
add_library(task1_nav SHARED src/plugins.cpp)
target_include_directories(task1_nav PUBLIC include)
target_link_libraries(task1_nav ompl::ompl)
ament_target_dependencies(task1_nav rclcpp rclcpp_lifecycle nav2_core nav2_costmap_2d nav2_util dwb_core dwb_msgs nav_2d_msgs pluginlib robotx_task1_interfaces)
add_library(task1_rotate SHARED src/rotate.cpp)
ament_target_dependencies(task1_rotate rclcpp dwb_core dwb_critics pluginlib)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
pluginlib_export_plugin_description_file(dwb_core plugins.xml)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS task1_nav task1_rotate LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_rules src/test_rules.cpp)
  target_include_directories(test_rules PRIVATE include)
  target_link_libraries(test_rules ompl::ompl)
  add_test(NAME rule_geometry_and_search COMMAND test_rules)
  add_executable(test_hull src/test_hull.cpp)
  target_include_directories(test_hull PRIVATE include)
  ament_target_dependencies(test_hull nav2_costmap_2d)
  add_test(NAME full_hull_occupancy COMMAND test_hull)
  add_executable(test_traffic_layer src/test_traffic_layer.cpp)
  target_include_directories(test_traffic_layer PRIVATE include)
  ament_target_dependencies(test_traffic_layer nav2_costmap_2d nav2_util)
  add_test(NAME traffic_layer_preserves_other_obstacles COMMAND test_traffic_layer)
endif()
ament_package()
